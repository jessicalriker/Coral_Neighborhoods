# 1. Same-morphotaxon nearest neighbor tables
same_files <- tibble(
  file = c(
    "data/Nearest Neighbor/BA_NearSameTaxa.csv",
    "data/Nearest Neighbor/FR4_NearSameTaxa.csv",
    "data/Nearest Neighbor/LA_NearSameTaxa.csv"
  ),
  site = c("BA", "FR4", "LA")
)

same_taxa <- same_files %>%
  mutate(data = map(file, ~ read_csv(.x, show_col_types = FALSE))) %>%
  select(-file) %>%
  unnest(data) %>%
  filter(!is.na(IN_FID), !is.na(NEAR_FID), !is.na(NEAR_DIST), IN_FID != NEAR_FID) %>%
  group_by(site, IN_FID) %>%
  summarise(
    closest_same_dist = min(NEAR_DIST, na.rm = TRUE),
    .groups = "drop"
  )

# 2. Different-morphotaxon / multi-rank tables (standardizing FR4's PRED_DESC_ / PRED_DESC1)
read_diff_reef <- function(path, reef_name) {
  df <- read_csv(path, show_col_types = FALSE) %>%
    mutate(site = reef_name)

  if (reef_name == "FR4") {
    df %>%
      mutate(
        taxa_in = as.character(PRED_DESC_),
        taxa_near = as.character(PRED_DESC1)
      )
  } else {
    df %>%
      mutate(
        taxa_in = as.character(CLEANEDPRE),
        taxa_near = as.character(CLEANEDPRE_1)
      )
  }
}

diff_raw <- pmap_dfr(
  list(
    c(
      "data/Nearest Neighbor/BA_neardifftaxa.csv",
      "data/Nearest Neighbor/FR4_neardifftaxa.csv",
      "data/Nearest Neighbor/LA_neardifftaxa.csv"
    ),
    c("BA", "FR4", "LA")
  ),
  read_diff_reef
)

focal_info <- diff_raw %>%
  filter(!is.na(IN_FID), !is.na(taxa_in)) %>%
  group_by(site, IN_FID) %>%
  summarise(
    morphotaxon = first(taxa_in),
    colony_area = first(IN_AREA),
    .groups = "drop"
  )

diff_taxa <- diff_raw %>%
  filter(
    !is.na(IN_FID), !is.na(NEAR_FID), !is.na(NEAR_DIST),
    IN_FID != NEAR_FID,
    !is.na(taxa_in), !is.na(taxa_near),
    taxa_near != taxa_in
  ) %>%
  group_by(site, IN_FID) %>%
  summarise(
    closest_diff_dist = min(NEAR_DIST, na.rm = TRUE),
    closest_diff_taxa = taxa_near[which.min(NEAR_DIST)],
    .groups = "drop"
  )

colony_proximity <- focal_info %>%
  full_join(same_taxa, by = c("site", "IN_FID")) %>%
  full_join(diff_taxa, by = c("site", "IN_FID")) %>%
  pivot_longer(
    cols = c(closest_same_dist, closest_diff_dist),
    names_to = "neighbor_type",
    values_to = "nearest_dist"
  ) %>%
  mutate(
    neighbor_type = recode(
      neighbor_type,
      closest_same_dist = "Same Morphotaxon",
      closest_diff_dist = "Different Morphotaxon"
    )
  ) %>%
  filter(!is.na(nearest_dist))

# Summary stats by site and neighbor_type
prox_summary <- colony_proximity %>%
  group_by(site, neighbor_type) %>%
  summarise(
    n = n(),
    median_dist = median(nearest_dist, na.rm = TRUE),
    iqr_dist = IQR(nearest_dist, na.rm = TRUE),
    q25 = quantile(nearest_dist, 0.25, na.rm = TRUE),
    q75 = quantile(nearest_dist, 0.75, na.rm = TRUE),
    .groups = "drop"
  )

# Overall Wilcoxon rank-sum test: Same vs. Different morphotaxon
wilcox_overall <- wilcox.test(nearest_dist ~ neighbor_type, data = colony_proximity)

# Within-site Wilcoxon tests
wilcox_by_site <- colony_proximity %>%
  group_by(site) %>%
  summarise(
    p_value = wilcox.test(nearest_dist ~ neighbor_type)$p.value,
    w_stat = unname(wilcox.test(nearest_dist ~ neighbor_type)$statistic),
    .groups = "drop"
  )

# Size vs. distance correlation (Spearman) where colony_area is available
size_cor_by_site <- colony_proximity %>%
  filter(!is.na(colony_area), !is.na(nearest_dist)) %>%
  group_by(site, neighbor_type) %>%
  summarise(
    n = n(),
    rho = cor(colony_area, nearest_dist, method = "spearman"),
    p_value = cor.test(colony_area, nearest_dist, method = "spearman", exact = FALSE)$p.value,
    .groups = "drop"
  )
